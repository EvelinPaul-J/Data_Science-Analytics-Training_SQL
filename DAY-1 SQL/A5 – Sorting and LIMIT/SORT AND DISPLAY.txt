-- 46. Sort all books by price in ascending order
SELECT * FROM Books 
ORDER BY price ASC;

-- 47. Sort all books by price in descending order
SELECT * FROM Books 
ORDER BY price DESC;

-- 48. Sort books alphabetically by book name
SELECT * FROM Books 
ORDER BY book_name ASC;

-- 49. Sort books by category and then by price
SELECT * FROM Books 
ORDER BY category ASC, price ASC;

-- 50. Display the 3 most expensive books
SELECT * FROM Books 
ORDER BY price DESC LIMIT 3;

-- 51. Display the 3 cheapest books
SELECT * FROM Books 
ORDER BY price ASC LIMIT 3;

-- 52. Display the 5 books with the highest stock quantity
SELECT * FROM Books 
ORDER BY stock_quantity DESC LIMIT 5;

-- 53. Display the first 5 members alphabetically by name
SELECT * FROM Members

-- 54. Display the latest 5 borrow records

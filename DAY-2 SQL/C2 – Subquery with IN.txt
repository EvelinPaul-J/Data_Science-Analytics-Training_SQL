-- 7. Display books belonging to categories having more than one book
SELECT * FROM Books 
WHERE category IN (SELECT category FROM Books GROUP BY category HAVING COUNT(*) > 1);

-- 8. Display books written by authors who have written more than one book
SELECT * FROM Books 
WHERE author IN (SELECT author FROM Books GROUP BY author HAVING COUNT(*) > 1);

-- 9. Display books belonging to categories whose average price is greater than ₹500
SELECT * FROM 
Books WHERE category IN (SELECT category FROM Books GROUP BY category HAVING AVG(price) > 500);

-- 10. Display members who have borrowed books from the Technology category
SELECT * FROM Members 
WHERE member_id IN (SELECT member_id FROM Borrow WHERE book_id IN (SELECT book_id FROM Books WHERE category = 'Technology'));

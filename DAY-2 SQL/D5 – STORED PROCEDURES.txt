-- 45. Create a procedure named GetBooksByCategory that accepts a category as an IN parameter
DELIMITER //
CREATE PROCEDURE GetBooksByCategory(IN p_category VARCHAR(50))
BEGIN

END //

-- 48. Create a procedure named GetBooksAbovePrice that accepts a price and displays books above that price
DELIMITER //
CREATE PROCEDURE GetBooksAbovePrice(IN p_price DECIMAL(10,2))
BEGIN
    SELECT * FROM Books WHERE price > p_price;
END //
DELIMITER ;

-- 49. Create a procedure named GetMemberBorrowDetails that accepts a member_id
DELIMITER //
CREATE PROCEDURE GetMemberBorrowDetails(IN p_member_id INT)
BEGIN
    SELECT Books.book_name, Books.author, Borrow.borrow_date, Borrow.return_date
    FROM Books INNER JOIN Borrow ON Books.book_id = Borrow.book_id
    WHERE Borrow.member_id = p_member_id;
END //
DELIMITER ;

-- 50. Create a procedure named GetCategoryBooks that accepts a category and price limit
DELIMITER //
CREATE PROCEDURE GetCategoryBooks(IN p_category VARCHAR(50), IN p_price DECIMAL(10,2))
BEGIN
    SELECT * FROM Books WHERE category = p_category AND price <= p_price;
END //
DELIMITER ;
